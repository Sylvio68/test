ServerName mon_domaine

<IfModule mod_ssl.c>
<VirtualHost *:443>
	ServerAdmin		mon_domaineee.fr

	# Blocaged des robots, y compris les moteurs de recherche
	RewriteEngine		Off

	# Activation des règles de modsecurity
	SecRuleEngine		On

	# Chargement des modules (déjà activé via a2enmod)
#	LoadModule proxy_module modules/mod_proxy.so
#	LoadModule proxy_http_module modules/mod_proxy_http.so
#	LoadModule headers_module modules/mod_headers.so

	# Protection de l'URL côté client
	ProxyPreserveHost	On

	# Évite les intrusions et l'usage d'Apache comme proxy ouvert avec des sources externes en plus du backend
	ProxyRequests		Off

	# Refléter le schéma côté backend
	RequestHeader		set X-Forwarded-Proto	"https"
	RequestHeader		set X-Forwarded-Host	"mon_domaine"
	RequestHeader		set X-Forwarded-Port	"443"

	# Redirection vers les serveurs
	# Nextcloud
	ProxyPass		/nextcloud/ http://nextcloud.mon_domaine/nextcloud/
	ProxyPassreverse	/nextcloud/ http://nextcloud.mon_domaine/nextcloud/

	# Protection des entêtes des URLs sources
	<Location /mon_service/>
		Header edit Location "^/$" /nextcloud/
		Header edit Location "^/nextcloud/?$" "/nextcloud/"
	</Location>

	# Joomla
	ProxyPass		/mon_service2/ http://mon_service2.mon_domaine/mon_service2/
	ProxyPassReverse	/mon_service2/ http://mon_service2.mon_domaine/mon_service2/

	# Protection des entêtes des URLs sources
	<Location /mon_service2/>
		Header edit Location "^/$" /mon_service2/
		Header edit Location "^/mon_service2/?$" "/mon_service2/"
	</Location>

Côté Nextcloud :

<VirtualHost *:80>

	ServerAdmin admin@localhost

	DocumentRoot /var/www/html/

#	Redirect permanent /nextcloud /
	Alias /nextcloud /var/www/html/nextcloud

	<Location /var/www/html/nextcloud/>
		<LimitExcept GET POST>
			Order Deny,Allow
			Deny from all
		</LimitExcept>
	</Location>

	<Directory /var/www/html/nextcloud/>
		Options +FollowSymlinks
		AllowOverride All

		<IfModule mod_dav.c>
			Dav Off
		</IfModule>

		SetEnv HOME /var/www/html/nextcloud
		SetEnv HTTP_HOME /var/www/html/nextcloud
	</Directory>

#	<Location />
#		Require all denied
#	</Location>

	# Mise en cache pour accélérer le fonctionnement du site
	<FilesMatch "\.(html|css|js|jpg|png)$">
		Header set Cache-Control "max-age=604800, public"
	</FilesMatch>

	ErrorLog /var/log/apache2/nextcloud_error.log
	CustomLog /var/log/apache2/nextcloud_access.log combined

</VirtualHost>
